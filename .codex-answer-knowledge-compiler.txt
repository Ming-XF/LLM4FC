class PatchEmbedding(nn.Module):
    """Time-LLM patching: replicate-pad, unfold, project each patch to d_model.

    Mirrors the upstream Time-LLM PatchEmbedding (layers/Embed.py).
    """

    def __init__(self, d_model, patch_len, stride, dropout):
        super().__init__()
        self.patch_len = patch_len
        self.stride = stride
        self.padding_patch_layer = nn.ReplicationPad1d((0, stride))
        self.value_embedding = nn.Conv1d(
            patch_len, d_model, kernel_size=3,
            padding=1, padding_mode='circular', bias=False)
        nn.init.kaiming_normal_(
            self.value_embedding.weight, mode='fan_in',
            nonlinearity='leaky_relu')
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # x: (B, N, T) raw EEG
        n_vars = x.shape[1]
        x = self.padding_patch_layer(x)
        x = x.unfold(dimension=-1, size=self.patch_len, step=self.stride)
        x = torch.reshape(x, (x.shape[0] * x.shape[1], x.shape[2], x.shape[3]))
        x = self.value_embedding(x.permute(0, 2, 1)).transpose(1, 2)
        return self.dropout(x), n_vars

class ReprogrammingLayer(nn.Module):
    def __init__(self, d_model, n_heads, d_keys=None, d_llm=None,
                 attention_dropout=0.1):
        super().__init__()
        d_keys = d_keys or (d_model // n_heads)

        self.query_projection = nn.Linear(d_model, d_keys * n_heads)
        self.key_projection = nn.Linear(d_llm, d_keys * n_heads)
        self.value_projection = nn.Linear(d_llm, d_keys * n_heads)
        self.out_projection = nn.Linear(d_keys * n_heads, d_llm)
        self.n_heads = n_heads
        self.dropout = nn.Dropout(attention_dropout)

    def forward(self, target_embedding, source_embedding, value_embedding):
        B, L, _ = target_embedding.shape
        S, _ = source_embedding.shape
        H = self.n_heads

        target_embedding = self.query_projection(target_embedding).view(
            B, L, H, -1)
        source_embedding = self.key_projection(source_embedding).view(S, H, -1)
        value_embedding = self.value_projection(value_embedding).view(S, H, -1)

        out = self._reprogramming(
            target_embedding, source_embedding, value_embedding)
        return self.out_projection(out.reshape(B, L, -1))

    def _reprogramming(self, target_embedding, source_embedding,
                       value_embedding):
        _, _, _, E = target_embedding.shape
        scale = 1. / sqrt(E)
        scores = torch.einsum(
            "blhe,she->bhls", target_embedding, source_embedding)
        attention = self.dropout(torch.softmax(scale * scores, dim=-1))
        return torch.einsum(
            "bhls,she->blhe", attention, value_embedding)

def _build_edge_masks(C, device):
    """构建 (C, C) 布尔上三角边类型掩码集合。"""
    triu = torch.triu(torch.ones(C, C, device=device, dtype=torch.bool),
                      diagonal=1)
    L = torch.zeros(C, device=device, dtype=torch.bool)
    R = torch.zeros(C, device=device, dtype=torch.bool)
    Fm = torch.zeros(C, device=device, dtype=torch.bool)
    P = torch.zeros(C, device=device, dtype=torch.bool)
    L[_CH_LEFT] = True
    R[_CH_RIGHT] = True
    Fm[_CH_FRONTAL] = True
    P[_CH_POSTERIOR] = True
    sides = L | R                                  # 排除中线
    inter = triu & ((L[:, None] & R[None, :]) | (R[:, None] & L[None, :]))
    intra = triu & sides[:, None] & sides[None, :] & ~inter
    frontal = triu & Fm[:, None] & Fm[None, :]
    posterior = triu & P[:, None] & P[None, :]
    return inter, intra, frontal, posterior


def _quantize(x, thresholds, label_list):
    """按阈值把 (B,) 张量离散为标签词列表。thresholds: [t1, t2]（升序）。"""
    idx = torch.zeros_like(x, dtype=torch.long)
    for t in thresholds:
        idx += (x >= t).long()
    return [label_list[i] for i in idx.tolist()]


def compute_graph_evidence(dfc, spectral=None, morph=None):
    """对 (B, W, C, C) DFC 计算确定性图统计量并量化为证据描述。

    Args:
        dfc: (B, W, C, C) 动态功能连接。
        spectral: (B, 9) 频谱统计量（数据集层确定性计算），None 时
            仅用 FC 证据（向后兼容）。
        morph: (B, 14) 形态学/周期-非周期统计量（CAER-v3），None 时
            输出保持 18 维（v2 兼容）。

    Returns:
        levels: (B,) list，每个元素为证据词列表
            （FC 6 句 + 频谱 4 句 [+ 形态 2 句]）。
        stats:  (B, 9) / (B, 18) / (B, 32) 连续统计量（裁剪 ±5）。
    """
    B, W, C, _ = dfc.shape
    device = dfc.device

    with torch.no_grad():
        abs_dfc = dfc.abs()
        # 全矩阵强度（含对角：对角=1 作为每窗口的锚点参照，
        # 使统计量跨站点更可比——见 WORK.md 诊断）
        win_strength = abs_dfc.mean(dim=(-1, -2))                       # (B, W)
        strength = win_strength.mean(-1)                                # (B,)
        variability = win_strength.std(-1)                              # (B,)
        # 时间趋势：每窗口强度对窗口序号的归一化斜率
        t = torch.arange(W, device=device).float()
        t = (t - t.mean())
        slope = (t * (win_strength - win_strength.mean(-1, keepdim=True))
                 ).sum(-1) / (t ** 2).sum()
        trend_norm = slope / (win_strength.mean(-1) + 1e-6)

        # 子矩阵均值（含对角，与跨域诊断的 old-def 一致）
        abs_all = abs_dfc.sum(dim=(-1, -2))                             # (B, W)

        def sub_sum(idx):
            return abs_dfc[:, :, idx][:, :, :, idx].sum(dim=(-1, -2))

        left_s = sub_sum(_CH_LEFT)
        right_s = sub_sum(_CH_RIGHT)
        front_s = sub_sum(_CH_FRONTAL)
        post_s = sub_sum(_CH_POSTERIOR)
        cent_s = sub_sum(_CH_CENTRAL)
        temp_s = sub_sum(_CH_TEMPORAL)
        pari_s = sub_sum(_CH_PARIETAL)
        occ_s = sub_sum(_CH_OCCIPITAL)
        mid_s = sub_sum(_CH_MIDLINE)
        inter_w = (abs_all - left_s - right_s) / (
            2 * len(_CH_LEFT) * len(_CH_RIGHT))
        intra_w = (left_s + right_s) / (
            len(_CH_LEFT) ** 2 + len(_CH_RIGHT) ** 2)
        front_w = front_s / len(_CH_FRONTAL) ** 2
        post_w = post_s / len(_CH_POSTERIOR) ** 2
        asym_w = (left_s / len(_CH_LEFT) ** 2
                  - right_s / len(_CH_RIGHT) ** 2).abs()

        inter_ratio = inter_w.mean(-1) / (intra_w.mean(-1) + 1e-6)
        front_ratio = front_w.mean(-1) / (post_w.mean(-1) + 1e-6)
        asym = asym_w.mean(-1)
        cent_mw = (cent_s / len(_CH_CENTRAL) ** 2).mean(-1)
        temp_mw = (temp_s / len(_CH_TEMPORAL) ** 2).mean(-1)
        pari_mw = (pari_s / len(_CH_PARIETAL) ** 2).mean(-1)
        occ_mw = (occ_s / len(_CH_OCCIPITAL) ** 2).mean(-1)
        mid_mw = (mid_s / len(_CH_MIDLINE) ** 2).mean(-1)

        # ── 量化（相关值是尺度无关的，固定阈值跨站点可解释）──
        strength_labels = _quantize(
            strength, [0.15, 0.30], ['low', 'mid', 'high'])
        inter_labels = _quantize(
            inter_ratio, [0.7, 1.0], ['weak', 'mid', 'strong'])
        front_labels = _quantize(
            front_ratio, [0.7, 1.0], ['low', 'mid', 'high'])
        var_labels = _quantize(
            variability, [0.02, 0.05], ['low', 'mid', 'high'])
        asym_labels = _quantize(
            asym, [0.02, 0.05], ['small', 'mid', 'large'])
        trend_idx = torch.where(
            trend_norm > 0.02, torch.full_like(trend_norm, 2, dtype=torch.long),
            torch.where(trend_norm < -0.02,
                        torch.full_like(trend_norm, 0, dtype=torch.long),
                        torch.ones_like(trend_norm, dtype=torch.long)))
        trend_words = ['down', 'flat', 'up']

    levels = []
    for b in range(B):
        levels.append([
            strength_labels[b], inter_labels[b], front_labels[b],
            var_labels[b], trend_words[int(trend_idx[b])], asym_labels[b],
        ])

    # 连续统计量按名义尺度重缩放（~单位量级，供线性头学习）；
    # 裁剪重尾。FC 9 维：4 个跨域一致的核心特征 + 5 个脑区均值
    # （variability/trend 两特征在 CAUEEG 与 TUAB 的年龄相关性符号
    # 相反，已剔除——见 WORK.md 诊断；hub-ness 同样破坏迁移，已剔除）
    fc_stats = torch.stack([
        strength / 0.3,
        inter_ratio,
        front_ratio,
        asym / 0.05,
        cent_mw / 0.3,
        temp_mw / 0.3,
        pari_mw / 0.3,
        occ_mw / 0.3,
        mid_mw / 0.3,
    ], dim=-1)                                                 # (B, 9)
    fc_stats = fc_stats.clamp(-5.0, 5.0)

    if spectral is None:
        return levels, fc_stats

    # ── 频谱证据（CAER-v2）：量化 4 个频谱句 + 9 维连续频谱统计量 ──
    spec = spectral.float().to(device)
    rel_d, rel_t, rel_a, rel_b = spec[:, 0], spec[:, 1], spec[:, 2], spec[:, 3]
    tb_ratio, a_dt, slowing, fa_pa, tt_total = (spec[:, 4], spec[:, 5],
                                                spec[:, 6], spec[:, 7],
                                                spec[:, 8])
    with torch.no_grad():
        slowing_labels = _quantize(
            slowing, [0.6, 1.2], ['low', 'mid', 'high'])
        alpha_labels = _quantize(
            rel_a, [0.25, 0.5], ['low', 'mid', 'high'])
        tb_labels = _quantize(
            tb_ratio, [0.8, 1.6], ['low', 'mid', 'high'])
        fb_idx = torch.where(
            fa_pa < 0.8, torch.zeros_like(fa_pa, dtype=torch.long),
            torch.where(fa_pa > 1.3, torch.full_like(fa_pa, 2, dtype=torch.long),
                        torch.ones_like(fa_pa, dtype=torch.long)))
        fb_words = ['front', 'even', 'back']
    for b in range(B):
        levels[b].extend([
            slowing_labels[b], alpha_labels[b], tb_labels[b],
            fb_words[int(fb_idx[b])],
        ])

    spec_stats = spec.clamp(-5.0, 5.0)
    stats = torch.cat([fc_stats, spec_stats], dim=-1)           # (B, 18)

    if morph is None:
        return levels, stats

    # ── 形态学证据（CAER-v3）：2 句证据文本 + 14 维连续统计量 ──
    # 顺序: ll_mean ll_max sharp_mean sharp_max sharp_focal kurt_mean
    #       kurt_max zcr_mean gamma_pow spike_mean spike_max apf apf_std slope
    m = morph.float().to(device)
    with torch.no_grad():
        sharp_labels = _quantize(
            m[:, 3], [6.0, 15.0], ['low', 'mid', 'high'])
        apf_idx = torch.where(
            m[:, 12] < 0.3, torch.zeros_like(m[:, 12], dtype=torch.long),
            torch.where(m[:, 12] > 0.6,
                        torch.full_like(m[:, 12], 2, dtype=torch.long),
                        torch.ones_like(m[:, 12], dtype=torch.long)))
        apf_words = ['smooth', 'mid', 'sharp']
    for b in range(B):
        levels[b].extend([
            sharp_labels[b], apf_words[int(apf_idx[b])],
        ])

    # 按各特征典型量级缩放（~单位量级供线性头学习），裁剪重尾
    _MORPH_SCALES = torch.tensor(
        [0.3, 0.5, 0.4, 15.0, 4.0, 4.0, 10.0, 0.12, 0.03,
         0.01, 0.03, 9.5, 0.5, 1.5], device=device)
    morph_stats = (m / _MORPH_SCALES).clamp(-5.0, 5.0)
    stats = torch.cat([stats, morph_stats], dim=-1)             # (B, 32)
    return levels, stats

def get_stats_schema(stats_dim):
    """按统计量宽度返回有序 (分组头, 字段表) 列表。

    stats_dim: 9（仅 FC）/ 18（FC+频谱）/ 32（全三族）。
    """
    if stats_dim == 32:
        return [
            (_FC_GROUP, _FC_FIELDS),
            (_SPEC_GROUP, _SPEC_FIELDS),
            (_MORPH_GROUP, _MORPH_FIELDS),
        ]
    if stats_dim == 18:
        return [(_FC_GROUP, _FC_FIELDS), (_SPEC_GROUP, _SPEC_FIELDS)]
    if stats_dim == 9:
        return [(_FC_GROUP, _FC_FIELDS)]
    raise ValueError(f"Unexpected stats dim {stats_dim}")


def _format_value(v):
    """固定 4 位小数、强制正负号、无科学计数法；-0.0000 归正。"""
    v = float(v)
    if not math.isfinite(v):
        v = 0.0
    v = max(-5.0, min(5.0, v))
    s = f"{v:+.4f}"
    return "+0.0000" if s == "-0.0000" else s


def build_stats_text(stats):
    """stats: (B, D) display 值 → 每样本一段统计 prompt 文本。"""
    groups = get_stats_schema(stats.shape[-1])
    texts = []
    for row in stats:
        parts = [_STATS_HEADER]
        offset = 0
        for group_name, fields in groups:
            parts.append(group_name)
            for i, (key, name) in enumerate(fields):
                parts.append(f"{name} [{key}] = {_format_value(row[offset + i])}")
            offset += len(fields)
        parts.append(_STATS_END)
        texts.append("\n".join(parts))
    return texts

    @staticmethod
    def spectral_stats(time_series, hz):
        """CAER-v2: 确定性频谱证据（9 维，无监督规则）。

        对 10 个滑窗（12s）做 Welch PSD（2s 分段、hanning 窗），
        取全通道均值得到 delta/theta/alpha/beta 相对功率，以及
        theta/beta、alpha/(d+t)、慢波指数 (d+t)/(a+b)、
        额/枕 alpha 比、颞区 theta 占比。

        Returns:
            torch.Tensor shape (9,)
        """
        W = 12 * hz
        step = (60 * hz - W) // 9
        nperseg = 2 * hz
        freqs = np.fft.rfftfreq(nperseg, 1 / hz)
        bands = {'d': (0.5, 4.0), 't': (4.0, 8.0),
                 'a': (8.0, 13.0), 'b': (13.0, 30.0)}
        frontal = [0, 1, 5, 6, 10, 13]
        posterior = [3, 4, 8, 9, 12, 15]
        temporal = [11, 12, 14, 15]
        win = np.hanning(nperseg)
        rows = []
        ts = time_series.detach().numpy().astype(np.float64)
        nseg = W // nperseg
        for k in range(10):
            s = k * step
            w = ts[:, s:s + W]
            w = (w - w.mean(1, keepdims=True)) / (w.std(1, keepdims=True) + 1e-8)
            pw = {bn: [] for bn in bands}
            for ch in range(w.shape[0]):
                seg = w[ch, :nseg * nperseg].reshape(nseg, nperseg)
                ps = (np.abs(np.fft.rfft(seg * win, axis=1)) ** 2).mean(0)
                for bn, (lo, hi) in bands.items():
                    pw[bn].append(ps[(freqs >= lo) & (freqs < hi)].sum())
            total = sum(np.mean(v) for v in pw.values())
            rel = {bn: np.mean(v) / total for bn, v in pw.items()}
            fa = sum(pw['a'][ch] for ch in frontal)
            pa = sum(pw['a'][ch] for ch in posterior)
            tt = sum(pw['t'][ch] for ch in temporal)
            rows.append([
                rel['d'], rel['t'], rel['a'], rel['b'],
                rel['t'] / max(rel['b'], 1e-9),
                rel['a'] / max(rel['d'] + rel['t'], 1e-9),
                (rel['d'] + rel['t']) / max(rel['a'] + rel['b'], 1e-9),
                fa / max(pa, 1e-9),
                tt / total,
            ])
        return torch.tensor(np.array(rows).mean(0), dtype=torch.float32)

    @staticmethod
    def morph_stats(time_series, hz):
        """CAER-v3: 确定性形态学/周期-非周期证据（14 维，无监督规则）。

        时域形态（发作间期放电相关）:
          ll_mean/ll_max      线长（|Δx| 均值，逐通道均值/最大）
          sharp_mean/max/focal 锐度 |ΔΔx|（均值/最大/跨通道灶性 std）
          kurt_mean/max       幅度峰度（瞬态尖波的厚尾特征）
          zcr_mean            零交叉率
          gamma_pow           30-45Hz 相对功率
          spike_mean/max      |ΔΔx|>6×MAD 的棘波速率（均值/最大通道）
        周期-非周期（年龄相关）:
          apf / apf_std       alpha 峰值频率（7-13Hz，均值/跨通道 std）
          slope               2-40Hz log-log PSD 非周期斜率

        Returns:
            torch.Tensor shape (14,)
        """
        ts = time_series.detach().numpy().astype(np.float64)
        ts = (ts - ts.mean(1, keepdims=True)) / (ts.std(1, keepdims=True) + 1e-8)
        C, N = ts.shape
        W = 12 * hz
        step = (60 * hz - W) // 9

        d1 = np.abs(np.diff(ts, axis=1))
        d2 = np.abs(np.diff(ts, n=2, axis=1))
        ll = d1.sum(1) / (N - 1)
        sharp = d2.mean(1)
        sharp_max_ch = d2.max(1)
        kurt = ((ts - ts.mean(1, keepdims=True)) ** 4).mean(1) / (
            (ts ** 2).mean(1) ** 2 + 1e-12) - 3.0
        zcr = (np.diff(ts > 0, axis=1) != 0).sum(1) / (N - 1)
        mad = np.median(d2, axis=1) / 0.6745
        spike = ((d2 > 6 * mad[:, None]).sum(1)) / (N - 2)

        # PSD: 12s 窗 × 4s 分段 → 0.25Hz 分辨率
        nperseg = 4 * hz
        nseg = W // nperseg
        freqs = np.fft.rfftfreq(nperseg, 1 / hz)
        win = np.hanning(nperseg)

class PatchEmbedding(nn.Module):
    """Time-LLM patching: replicate-pad, unfold, project each patch to d_model.

    Mirrors the upstream Time-LLM PatchEmbedding (layers/Embed.py).
    """

    def __init__(self, d_model, patch_len, stride, dropout):
        super().__init__()
        self.patch_len = patch_len
        self.stride = stride
        self.padding_patch_layer = nn.ReplicationPad1d((0, stride))
        self.value_embedding = nn.Conv1d(
            patch_len, d_model, kernel_size=3,
            padding=1, padding_mode='circular', bias=False)
        nn.init.kaiming_normal_(
            self.value_embedding.weight, mode='fan_in',
            nonlinearity='leaky_relu')
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        # x: (B, N, T) raw EEG
        n_vars = x.shape[1]
        x = self.padding_patch_layer(x)
        x = x.unfold(dimension=-1, size=self.patch_len, step=self.stride)
        x = torch.reshape(x, (x.shape[0] * x.shape[1], x.shape[2], x.shape[3]))
        x = self.value_embedding(x.permute(0, 2, 1)).transpose(1, 2)
        return self.dropout(x), n_vars

class ReprogrammingLayer(nn.Module):
    def __init__(self, d_model, n_heads, d_keys=None, d_llm=None,
                 attention_dropout=0.1):
        super().__init__()
        d_keys = d_keys or (d_model // n_heads)

        self.query_projection = nn.Linear(d_model, d_keys * n_heads)
        self.key_projection = nn.Linear(d_llm, d_keys * n_heads)
        self.value_projection = nn.Linear(d_llm, d_keys * n_heads)
        self.out_projection = nn.Linear(d_keys * n_heads, d_llm)
        self.n_heads = n_heads
        self.dropout = nn.Dropout(attention_dropout)

    def forward(self, target_embedding, source_embedding, value_embedding):
        B, L, _ = target_embedding.shape
        S, _ = source_embedding.shape
        H = self.n_heads

        target_embedding = self.query_projection(target_embedding).view(
            B, L, H, -1)
        source_embedding = self.key_projection(source_embedding).view(S, H, -1)
        value_embedding = self.value_projection(value_embedding).view(S, H, -1)

        out = self._reprogramming(
            target_embedding, source_embedding, value_embedding)
        return self.out_projection(out.reshape(B, L, -1))

    def _reprogramming(self, target_embedding, source_embedding,
                       value_embedding):
        _, _, _, E = target_embedding.shape
        scale = 1. / sqrt(E)
        scores = torch.einsum(
            "blhe,she->bhls", target_embedding, source_embedding)
        attention = self.dropout(torch.softmax(scale * scores, dim=-1))
        return torch.einsum(
            "bhls,she->blhe", attention, value_embedding)

    def forward(self, x, labels):
        B, N, T = x.shape
        device = x.device

        # Per-sample per-channel z-score: amplitude invariance is the key
        # property for cross-domain zero-shot transfer.
        mean = x.mean(dim=-1, keepdim=True)
        std = x.std(dim=-1, keepdim=True)
        x = (x - mean) / (std + 1e-5)

        # Patch embedding, then channel-major flatten:
        # [c0p0, ..., c0p(P-1), c1p0, ..., c(N-1)p(P-1)].
        x = x.to(dtype=self.patch_embedding.value_embedding.weight.dtype)
        enc_out, _ = self.patch_embedding(x)
        P = enc_out.shape[1]
        enc_out = enc_out.reshape(B, N * P, -1)

        word_embeddings = self.word_embeddings.permute(1, 0).to(
            dtype=self.mapping_layer.weight.dtype)
        source_embeddings = self.mapping_layer(
            word_embeddings).permute(1, 0)
        source_embeddings = source_embeddings.to(
            dtype=self.word_embeddings.dtype)

        reprogram_dtype = \
            self.reprogramming_layer.query_projection.weight.dtype
        reprogrammed = self.reprogramming_layer(
            enc_out.to(dtype=reprogram_dtype),
            source_embeddings.to(dtype=reprogram_dtype),
            source_embeddings.to(dtype=reprogram_dtype),
        ).to(dtype=torch.bfloat16)

        start_tag = self.start_tag_embeddings.unsqueeze(0).expand(B, -1, -1)
        end_tag = self.end_tag_embeddings.unsqueeze(0).expand(B, -1, -1)
        task_prompt = self.task_prompt_embeddings.unsqueeze(0).expand(
            B, -1, -1)

        prompt_parts = [start_tag]
        prompt_length = self.P_start
        if self.config.use_task_prompt:
            prompt_parts.append(task_prompt)
            prompt_length += self.P_task
        prompt_parts.append(end_tag)
        prompt_length += self.P_end
        prompt_parts.append(reprogrammed.to(dtype=task_prompt.dtype))

        inputs_embeds = torch.cat(prompt_parts, dim=1)
        sequence_length = inputs_embeds.shape[1]

        if self.llm_type == 'chatglm':
            causal_mask = torch.triu(
                torch.ones(
                    sequence_length, sequence_length,
                    dtype=torch.bool, device=device),
                diagonal=1,
            )
            attention_mask = causal_mask.unsqueeze(0).unsqueeze(0).expand(
                B, 1, -1, -1)
            global_pos = torch.arange(
                sequence_length, dtype=torch.long, device=device
            ).unsqueeze(0).repeat(B, 1)
            if self.position_encoding_2d:
                block_pos = torch.cat([
                    torch.arange(
                        prompt_length, dtype=torch.long, device=device),
                    torch.arange(N * P, dtype=torch.long, device=device),
                ]).unsqueeze(0).repeat(B, 1)
                position_ids = torch.stack((global_pos, block_pos), dim=1)
            else:
                position_ids = global_pos

            transformer_outputs = self._transformer(
                input_ids=None,
                position_ids=position_ids,
                attention_mask=attention_mask,
                past_key_values=None,
                inputs_embeds=inputs_embeds,
                use_cache=False,
                output_attentions=False,
                output_hidden_states=False,
                return_dict=False,
            )
            hidden = transformer_outputs[0].transpose(0, 1)
        else:
            position_ids = torch.arange(
                sequence_length, dtype=torch.long, device=device
            ).unsqueeze(0).expand(B, -1)
            transformer_outputs = self._transformer(
                input_ids=None,
                position_ids=position_ids,
                attention_mask=None,
                past_key_values=None,
                inputs_embeds=inputs_embeds,
                use_cache=False,
                output_attentions=False,
                output_hidden_states=False,
                return_dict=False,
            )
            hidden = transformer_outputs[0]

        hidden_patches = hidden[:, prompt_length:, :self.config.d_ff].to(
            dtype=self.output_projection[1].weight.dtype)

        # (B, N*P, d_ff) -> (B, N, P, d_ff) -> channel mean-pool ->
        # (B, P, d_ff) -> flatten -> scalar.
        hidden_patches = hidden_patches.reshape(
            B, N, self.config.patch_nums, -1)
        hidden_patches = hidden_patches.mean(dim=1)
        logits = self.output_projection(hidden_patches)

        if self.task_type == 'regression':
            prediction = logits.squeeze(-1)
            if labels.dim() > 1 and labels.shape[-1] == 1:

                prediction, labels.float(), delta=1.0)

            logits=logits,
            loss=loss,
            hidden_state={

                'reprogrammed': reprogrammed,
                'HL_patches': hidden_patches,
